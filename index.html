        window.calculateBSAandDoses = function() {

            if (!isNaN(h) && !isNaN(w) && h > 0 && w > 0) {

                bsa = Math.sqrt((h * w) / 3600);

                bsaDisplay.innerText = bsa.toFixed(2) + " m²";

            } else {

                bsaDisplay.innerText = "--";

        window.renderMarkerChart = function() {

            const p = window.currentViewingPatient;

            if (!p) return;


            const selectedType = document.getElementById('chartMarkerSelect').value;

            const markers = p.markers || [];

            

            const filtered = markers

                .filter(m => m.type === selectedType)

                .sort((a, b) => new Date(a.date) - new Date(b.date));


            const card = document.getElementById('markerAnalysisCard');

            const tableBody = document.getElementById('markerDetailTableBody');

            tableBody.innerHTML = "";


            if (filtered.length === 0) {

                card.innerHTML = `<span class="text-slate-500 italic"><i class="fa-solid fa-circle-info"></i> Chưa có dữ liệu chỉ số <strong>${selectedType}</strong> được ghi nhận.</span>`;

                    window.activeMarkerChart.destroy();

                    window.activeMarkerChart = null;

                }

                return;

            }


            let analysisText = "";

            if (filtered.length === 1) {

                analysisText = `<span>Chỉ số <strong>${selectedType}</strong> đo ngày ${filtered[0].date}: <strong>${filtered[0].value}</strong> (Cần thêm dữ liệu lần sau để so sánh %).</span>`;

            } else {

                const prev = parseFloat(filtered[filtered.length - 2].value);

                const curr = parseFloat(filtered[filtered.length - 1].value);

                const diff = curr - prev;

                const percent = ((diff / prev) * 100).toFixed(1);


                if (diff > 0) {

                    analysisText = `<span class="text-rose-700 flex items-center gap-1.5"><i class="fa-solid fa-arrow-trend-up text-base"></i> Chỉ số <strong>${selectedType}</strong> TĂNG <strong>+${percent}%</strong> (từ ${prev} lên ${curr}) so với lần đo trước (${filtered[filtered.length - 2].date}).</span>`;

                } else if (diff < 0) {

                    analysisText = `<span class="text-emerald-700 flex items-center gap-1.5"><i class="fa-solid fa-arrow-trend-down text-base"></i> Chỉ số <strong>${selectedType}</strong> GIẢM <strong>${percent}%</strong> (từ ${prev} xuống ${curr}) so với lần đo trước (${filtered[filtered.length - 2].date}).</span>`;

                    analysisText = `<span class="text-slate-700 flex items-center gap-1.5"><i class="fa-solid fa-minus text-base"></i> Chỉ số <strong>${selectedType}</strong> KHÔNG ĐỔI (${curr}) so với lần đo trước.</span>`;

                }

            }

            card.innerHTML = analysisText;


            filtered.forEach((item, index) => {

                let percentText = "-";

                if (index > 0) {

                    const prevVal = parseFloat(filtered[index - 1].value);

                    const currVal = parseFloat(item.value);

                    const pct = (((currVal - prevVal) / prevVal) * 100).toFixed(1);

                    if (pct > 0) percentText = `<span class="text-rose-600 font-bold">+${pct}% </span>`;

                    else if (pct < 0) percentText = `<span class="text-emerald-600 font-bold">${pct}% </span>`;

                    else percentText = `<span class="text-slate-500">0%</span>`;

                }


                const tr = document.createElement('tr');

                tr.innerHTML = `

                    <td class="p-2 font-medium">${item.date}</td>

                    <td class="p-2 font-bold text-indigo-900">${item.type}</td>

                    <td class="p-2 font-bold">${item.value}</td>

                    <td class="p-2">${percentText}</td>

                `;

                tableBody.appendChild(tr);

            });


            const ctx = document.getElementById('markerCanvas').getContext('2d');

            if (window.activeMarkerChart) {

                window.activeMarkerChart.destroy();

            }


            const labels = filtered.map(m => m.date);

            const dataValues = filtered.map(m => parseFloat(m.value));


            window.activeMarkerChart = new Chart(ctx, {

                type: 'line',

                data: {

                    labels: labels,

                    datasets: [{

                        label: `Chỉ số ${selectedType}`,

                        data: dataValues,

                        borderColor: '#4f46e5',

                        backgroundColor: 'rgba(79, 70, 229, 0.1)',

                        fill: true,

                        tension: 0.3,

                        pointRadius: 5,

                        pointHoverRadius: 7

                    }]

                },

                options: {

                    responsive: true,

                    maintainAspectRatio: false,

                    plugins: { legend: { display: true, position: 'top' } },

        window.runChemoStatistics = function() {

            const startDateStr = document.getElementById('statsStartDate').value;

            const endDateStr = document.getElementById('statsEndDate').value;

            const icdFilterVal = document.getElementById('statsIcdFilter').value.trim().toUpperCase();

            

            if (!startDateStr || !endDateStr) {

                alert("Vui lòng chọn đầy đủ Từ ngày và Đến ngày!");

                return;

            }


            const startDate = new Date(startDateStr);

            const endDate = new Date(endDateStr);

            endDate.setHours(23, 59, 59, 999);

            const currentYear = new Date().getFullYear();


            let matchedCount = 0;

            const statsTableBody = document.getElementById('statsTableBody');

            statsTableBody.innerHTML = "";


            let results = [];

            let genderCounts = { "Nam": 0, "Nữ": 0, "Khác": 0 };

            let ageCounts = { "Trẻ (< 35 tuổi)": 0, "Trung niên (35-60 tuổi)": 0, "Già (> 60 tuổi)": 0 };


            cachedPatientsList.forEach(p => {

                const diag = (p.diagnosis || "").toUpperCase();

                const subDiag = (p.subDiagnosis || "").toUpperCase();


                if (icdFilterVal !== "ALL") {

                    if (!diag.includes(icdFilterVal) && !subDiag.includes(icdFilterVal)) return;

                }


                const cycles = p.cycles || [];

                const validCycles = cycles.filter(c => {

                    if (!c.date) return false;

                    const cDate = new Date(c.date);

                    return cDate >= startDate && cDate <= endDate;

                });


                if (validCycles.length > 0 || cycles.length === 0) {

                    matchedCount++;

                    let g = p.gender || "Khác";

                    if (genderCounts[g] !== undefined) genderCounts[g]++;

                    else genderCounts["Khác"] = (genderCounts["Khác"] || 0) + 1;


                    let birthY = parseInt(p.birthYear) || (currentYear - 50);

                    let age = currentYear - birthY;

                    if (age < 35) ageCounts["Trẻ (< 35 tuổi)"]++;

                    else if (age <= 60) ageCounts["Trung niên (35-60 tuổi)"]++;

                    else ageCounts["Già (> 60 tuổi)"]++;


                    results.push({

                        id: p.id,

                        patientId: p.patientId || '-',

                        name: p.name || '-',

                        gender: p.gender || '-',

                        birthYear: p.birthYear || '-',

                        age: age,

                        diagnosis: p.diagnosis || '-',

                        subDiagnosis: p.subDiagnosis || 'Không có',

                        protocol: p.protocol || '-',

                        count: validCycles.length > 0 ? validCycles.length : cycles.length

                    });

                }

            });


            document.getElementById('statTotalPatients').innerText = matchedCount;

            document.getElementById('statGenderSummary').innerText = `Nam: ${genderCounts["Nam"]} | Nữ: ${genderCounts["Nữ"]}`;

            document.getElementById('statAgeSummary').innerText = `Trẻ: ${ageCounts["Trẻ (< 35 tuổi)"]} | Trung niên: ${ageCounts["Trung niên (35-60 tuổi)"]} | Già: ${ageCounts["Già (> 60 tuổi)"]}`;


            if (results.length === 0) {

                statsTableBody.innerHTML = `<tr><td colspan="8" class="p-6 text-center text-slate-400 font-medium">Không tìm thấy bệnh nhân nào trong khoảng thời gian này.</td></tr>`;

                return;

            }


            const canEdit = isAdmin || isVip;


            results.forEach(item => {

                const tr = document.createElement('tr');

                tr.className = "hover:bg-slate-50 transition";

                tr.innerHTML = `

                    <td class="p-3 font-bold text-teal-800">${item.patientId}</td>

                    <td class="p-3 font-semibold text-slate-900">${item.name}</td>

                    <td class="p-3 text-center">${item.gender}</td>

                    <td class="p-3">${item.birthYear} (${item.age}t)</td>

                    <td class="p-3 font-medium text-slate-800">

                        <strong class="text-teal-800">Chính:</strong> ${item.diagnosis}<br>

                        <strong class="text-amber-800">Phụ:</strong> ${item.subDiagnosis}

                    </td>

                    <td class="p-3 font-bold text-amber-800">${item.protocol}</td>

                    <td class="p-3 text-center font-bold text-teal-700">${item.count} chu kỳ</td>

                    <td class="p-3 text-center space-x-1">

                        <button onclick="window.viewPatientDetail('${item.id}')" title="Xem chi tiết" class="p-1.5 bg-teal-50 hover:bg-teal-100 text-teal-700 rounded"><i class="fa-solid fa-eye"></i></button>

                        ${canEdit ? `<button onclick="window.closeStatsModal(); window.openEditPatientModal('${item.id}');" title="Chỉnh sửa thông tin bệnh nhân" class="p-1.5 bg-amber-50 hover:bg-amber-100 text-amber-700 rounded"><i class="fa-solid fa-pen-to-square"></i> Sửa</button>` : ''}

                        ${isAdmin ? `<button onclick="window.deletePatientDoc('${item.id}'); window.runChemoStatistics();" title="Xóa" class="p-1.5 bg-rose-50 hover:bg-rose-100 text-rose-700 rounded"><i class="fa-solid fa-trash"></i></button>` : ''}

                    </td>

                `;

                statsTableBody.appendChild(tr);

            });

        };
